2. วิธีติดตั้ง Google Sheets 4 ขั้นตอน (ทำครั้งเดียวใช้ได้ทั้งเทอม)
  1. สร้างไฟล์ Google Sheets ใหม่ ตั้งชื่อว่า ฐานข้อมูลเก็บคะแนนเกม_ระบบสมการเชิงเส้นสองตัวแปร_ม3
  2. ไปที่เมนูด้านบนเลือก ส่วนขยาย (Extensions) > Apps Script
  3. ลบโค้ดเดิมทั้งหมด แล้วกดปุ่ม "คัดลอกโค้ด Apps Script (Code.gs)" ด้านบนมาวางแทนที่ จากนั้นกดบันทึก (💾)
  4. กดปุ่ม ทำให้ใช้งานได้ (Deploy) > การทำให้ใช้งานได้รายการใหม่ (New deployment) เลือกประเภทเป็น เว็บแอป (Web app) ตั้งค่าผู้มีสิทธิ์เข้าถึงเป็น ทุกคน (Everyone) กด Deploy แล้วนำ URL มาวางในช่องด้านบนครับ
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["วัน-เวลาที่ส่ง", "ชื่อ-นามสกุล", "ชื่อเล่น", "ชั้นเรียน", "เลขที่", "เส้นทางที่เลือก", "คะแนนด่าน1_ทบทวน", "คะแนนด่าน2_แบบทดสอบ", "คะแนนรวม", "ระดับผลการเรียน"]);
  }
  var data = JSON.parse(e.postData.contents);
  sheet.appendRow([
    new Date().toLocaleString("th-TH"),
    data.fullName,
    data.nickName,
    data.classroom,
    data.studentNo,
    data.pathLabel,
    data.warmupScore,
    data.quizScore,
    data.totalScore,
    data.gradeLabel
  ]);
  return ContentService.createTextOutput(JSON.stringify({status: "success"})).setMimeType(ContentService.MimeType.JSON);
}
